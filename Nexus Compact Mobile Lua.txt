local Connections = {}
local PlayerConnections = {}

local function track(conn)
	table.insert(Connections, conn)
	return conn
end

local function trackPlayer(player, conn)
	if not PlayerConnections[player] then
		PlayerConnections[player] = {}
	end
	table.insert(PlayerConnections[player], conn)
	return conn
end

local function cleanupPlayerConns(player)
	local list = PlayerConnections[player]
	if not list then
		return
	end
	for _, conn in ipairs(list) do
		if conn.Connected then
			conn:Disconnect()
		end
	end
	PlayerConnections[player] = nil
end

--========================================================

--========================================================

local function IsEnemy(player)
	if not player or player == LocalPlayer then
		return false
	end

	if LocalPlayer.Team ~= nil
		and player.Team ~= nil
		and LocalPlayer.Team == player.Team then
		return false
	end

	return true
end

local function GetAimPart(character)
	if not character then
		return nil
	end

	local part = character:FindFirstChild(Config.AimPart)
	if part and part:IsA("BasePart") then
		return part
	end

	-- Fallback between R15 UpperTorso and R6 Torso
	if Config.AimPart == "UpperTorso" then
		part = character:FindFirstChild("Torso")
	elseif Config.AimPart == "Torso" then
		part = character:FindFirstChild("UpperTorso")
	end

	if part and part:IsA("BasePart") then
		return part
	end

	return character:FindFirstChild("HumanoidRootPart")
		or character:FindFirstChild("Head")
end

		return

local function updateStatusText()
	local active = {}
	if Config.AimAssist then
		table.insert(active, "AIM")
	end
	if Config.ESP or Config.NameESP then
		table.insert(active, "ESP")
	end

	if #active > 0 then
		Subtitle.Text = "ACTIVE • " .. table.concat(active, " + ")
		AccentDot.BackgroundColor3 = Theme.Active
	else
		Subtitle.Text = "MOBILE • STANDBY"
		AccentDot.BackgroundColor3 = Theme.AccentBright
	end
end

	local button = Instance.new("TextButton")
	button.Size = UDim2.new(1, -2, 0, 40)
	button.BackgroundColor3 = Theme.Card
	button.Text = ""
	button.AutoButtonColor = true
	button.Parent = parent

	local corner = Instance.new("UICorner")

	corner.Parent = button

	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Position = UDim2.fromOffset(12, 0)
	label.Size = UDim2.new(1, -68, 1, 0)
	label.Text = title
	label.TextColor3 = Theme.Text
	label.TextSize = 11
	label.Font = Enum.Font.GothamBold
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = button

	badge.AnchorPoint = Vector2.new(1, 0.5)
	badge.Position = UDim2.new(1, -10, 0.5, 0)
	badge.Size = UDim2.fromOffset(44, 22)
	badge.TextSize = 10
	badge.Font = Enum.Font.GothamBold
	badge.BorderSizePixel = 0
	badge.Parent = button

	local badgeCorner = Instance.new("UICorner")
	badgeCorner.CornerRadius = UDim.new(0, 6)
	badgeCorner.Parent = badge

	local state = initialState

	local function refresh()
		badge.Text = state and "ON" or "OFF"
		badge.BackgroundColor3 = state and Theme.Active or Theme.Surface
		badge.TextColor3 = state and Theme.Bg or Theme.Subtext
	end

	button.Activated:Connect(function()
		state = not state
		refresh()
		onToggle(state)
		updateStatusText()
	end)

	refresh()
	return button, function(newState)
		state = newState
		refresh()
		onToggle(state)
		updateStatusText()
	end
end

	valueLabel.TextColor3 = Theme.AccentBright
	valueLabel.TextSize = 11
	valueLabel.Font = Enum.Font.GothamBold
	valueLabel.TextXAlignment = Enum.TextXAlignment.Right
	valueLabel.Parent = holder

	-- Touch-friendly 30px hitbox
	local hitbox = Instance.new("TextButton")
	hitbox.Position = UDim2.new(0, 12, 0, 21)
	hitbox.Size = UDim2.new(1, -24, 0, 28)
	hitbox.BackgroundTransparency = 1
	hitbox.Text = ""
	hitbox.AutoButtonColor = false
	hitbox.Parent = holder

	local trackBar = Instance.new("Frame")
	trackBar.Position = UDim2.new(0, 0, 0.5, -3)
	trackBar.Size = UDim2.new(1, 0, 0, 6)
	trackBar.BackgroundColor3 = Theme.Surface
	trackBar.BorderSizePixel = 0
	trackBar.Parent = hitbox

	local trackCorner = Instance.new("UICorner")
	trackCorner.CornerRadius = UDim.new(1, 0)
	trackCorner.Parent = trackBar

	local fill = Instance.new("Frame")
	fill.Size = UDim2.fromScale(0, 1)
	fill.BackgroundColor3 = Theme.Accent
	fill.BorderSizePixel = 0
	fill.Parent = trackBar

	local fillCorner = Instance.new("UICorner")
	fillCorner.CornerRadius = UDim.new(1, 0)
	fillCorner.Parent = fill

	local knob = Instance.new("Frame")
	knob.Size = UDim2.fromOffset(15, 15)
	knob.AnchorPoint = Vector2.new(0.5, 0.5)
	knob.Position = UDim2.fromScale(0, 0.5)
	knob.BackgroundColor3 = Theme.Text
	knob.BorderSizePixel = 0
	knob.Parent = trackBar

	local knobCorner = Instance.new("UICorner")
	knobCorner.CornerRadius = UDim.new(1, 0)
	knobCorner.Parent = knob

	local activeInput = nil

	local function setValue(v)
		local rounded = math.clamp(math.round(v), minValue, maxValue)
		local pct = (rounded - minValue) / math.max(1, maxValue - minValue)

		fill.Size = UDim2.fromScale(pct, 1)
		knob.Position = UDim2.fromScale(pct, 0.5)
		valueLabel.Text = tostring(rounded)
		onChange(rounded)
	end

	local function applyInput(input)
		local w = trackBar.AbsoluteSize.X
		if w <= 0 then
			return
		end
		local pct = math.clamp((input.Position.X - trackBar.AbsolutePosition.X) / w, 0, 1)
		setValue(minValue + (maxValue - minValue) * pct)
	end

	hitbox.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch
			or input.UserInputType == Enum.UserInputType.MouseButton1 then
			activeInput = input
			applyInput(input)
		end
	end)

	track(UserInputService.InputChanged:Connect(function(input)
		if not activeInput then
			return
		end
		if input == activeInput
			or (activeInput.UserInputType == Enum.UserInputType.MouseButton1
				and input.UserInputType == Enum.UserInputType.MouseMovement) then
			applyInput(input)
		end
	end))

	track(UserInputService.InputEnded:Connect(function(input)
		if input == activeInput then
			activeInput = nil
		end
	end))

	setValue(initialValue)
	return setValue
end

local ESPFolder = Instance.new("Folder")
ESPFolder.Name = ESP_NAME
ESPFolder.Parent = workspace

local function RemoveESP(player)
	local data = ESPObjects[player]
	if not data then
		return
	end

	if data.Highlight then
		data.Highlight:Destroy()
	end
	if data.NameTag then
		data.NameTag:Destroy()
	end

	ESPObjects[player] = nil
end

local function SyncPlayerESP(player)
	if not player or player == LocalPlayer then
		return
	end

	if (not Config.ESP and not Config.NameESP) or not IsEnemy(player) then
		RemoveESP(player)
		return
	end

	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local head = character and (character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart"))

	if not character or not humanoid or humanoid.Health <= 0 or not head then
		RemoveESP(player)
		return
	end

	local data = ESPObjects[player]
	if data and data.Character ~= character then
		RemoveESP(player)
		data = nil
	end

	if not data then
		data = { Character = character }
		ESPObjects[player] = data
	end

	-- Highlight ESP
	if Config.ESP and not data.Highlight then
		local highlight = Instance.new("Highlight")
		highlight.Name = "PlayerESP"
		highlight.Adornee = character
		highlight.FillColor = Theme.Accent
		highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
		highlight.FillTransparency = 0.65
		highlight.OutlineTransparency = 0
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		highlight.Parent = ESPFolder
		data.Highlight = highlight
	elseif not Config.ESP and data.Highlight then
		data.Highlight:Destroy()
		data.Highlight = nil
	end

	-- Name ESP
	if Config.NameESP and not data.NameTag then
		local tag = Instance.new("BillboardGui")
		tag.Name = "NameESP"
		tag.Adornee = head
		tag.Size = UDim2.fromOffset(180, 32)
		tag.StudsOffset = Vector3.new(0, 3, 0)
		tag.AlwaysOnTop = true
		tag.MaxDistance = Config.ESPDistance
		tag.Parent = ESPFolder

		local label = Instance.new("TextLabel")
		label.BackgroundTransparency = 1
		label.Size = UDim2.fromScale(1, 1)
		label.Text = player.DisplayName
		label.TextColor3 = Theme.Text
		label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
		label.TextStrokeTransparency = 0.2
		label.TextSize = 13
		label.Font = Enum.Font.GothamBold
		label.Parent = tag

		data.NameTag = tag
	elseif not Config.NameESP and data.NameTag then
		data.NameTag:Destroy()
		data.NameTag = nil
	end

	if data.NameTag then
		data.NameTag.Adornee = head
		data.NameTag.MaxDistance = Config.ESPDistance
	end
end

local function UpdateESP()
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			SyncPlayerESP(player)
		end
	end
end

--========================================================
-- TARGETING & AIM LOOP (CENTER-SCREEN MOBILE READY)
--========================================================

local function FindTarget(camera, center)
	local closest = nil
	local closestDistance = Config.FOV

	for _, player in ipairs(Players:GetPlayers()) do
		if IsEnemy(player) then
			local character = player.Character
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
			local part = GetAimPart(character)

			if humanoid and humanoid.Health > 0 and part then
				local screen, visible = camera:WorldToViewportPoint(part.Position)
				if visible and screen.Z > 0 then
					local distance = (Vector2.new(screen.X, screen.Y) - center).Magnitude
					if distance < closestDistance then
						closestDistance = distance
						closest = part
					end
				end
			end
		end
	end

	return closest
end

track(RunService.RenderStepped:Connect(function(dt)
	local camera = workspace.CurrentCamera
	if not camera then
		FOVCircle.Visible = false
		return
	end

	local center = Vector2.new(camera.ViewportSize.X * 0.5, camera.ViewportSize.Y * 0.5)
	FOVCircle.Position = UDim2.fromOffset(center.X, center.Y)
	FOVCircle.Size = UDim2.fromOffset(Config.FOV * 2, Config.FOV * 2)
	FOVCircle.Visible = Config.AimAssist

	if not Config.AimAssist then
		CurrentTarget = nil
		return
	end

	if CurrentTarget then
		local character = CurrentTarget.Parent
		local player = character and Players:GetPlayerFromCharacter(character)
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")

		if not CurrentTarget:IsDescendantOf(workspace)
			or not player
			or not IsEnemy(player)
			or not humanoid
			or humanoid.Health <= 0 then
			CurrentTarget = nil
		else
			local screen, visible = camera:WorldToViewportPoint(CurrentTarget.Position)
			if not visible or screen.Z <= 0
				or (Vector2.new(screen.X, screen.Y) - center).Magnitude > Config.FOV then
				CurrentTarget = nil
			end
		end
	end

	if not CurrentTarget then
		CurrentTarget = FindTarget(camera, center)
	end

	if not CurrentTarget then
		return
	end

	local desired = CFrame.lookAt(camera.CFrame.Position, CurrentTarget.Position)
	local alpha = 1 - math.exp(-Config.AimSpeed * dt)
	camera.CFrame = camera.CFrame:Lerp(desired, alpha)
end))

-- Lightweight ESP sync loop (every 0.2s)
local espTimer = 0
track(RunService.Heartbeat:Connect(function(dt)
	espTimer += dt
	if espTimer < 0.2 then
		return
	end
	espTimer = 0
	UpdateESP()
end))

--========================================================
-- PLAYER LIFECYCLE
--========================================================

local function SetupPlayer(player)
	if player == LocalPlayer then
		return
	end

	trackPlayer(player, player.CharacterAdded:Connect(function()
		RemoveESP(player)
	end))

	trackPlayer(player, player.CharacterRemoving:Connect(function(character)
		RemoveESP(player)
		if CurrentTarget and CurrentTarget:IsDescendantOf(character) then
			CurrentTarget = nil
		end
	end))

	trackPlayer(player, player:GetPropertyChangedSignal("Team"):Connect(function()
		if not IsEnemy(player) then
			RemoveESP(player)
			if CurrentTarget and player.Character and CurrentTarget:IsDescendantOf(player.Character) then
				CurrentTarget = nil
			end
		end
	end))
end

for _, player in ipairs(Players:GetPlayers()) do
	SetupPlayer(player)
end

track(Players.PlayerAdded:Connect(SetupPlayer))

track(Players.PlayerRemoving:Connect(function(player)
	RemoveESP(player)
	cleanupPlayerConns(player)
	if CurrentTarget and player.Character and CurrentTarget:IsDescendantOf(player.Character) then
		CurrentTarget = nil
	end
end))

--========================================================
-- TOUCH-FRIENDLY WINDOW DRAG
--========================================================

local dragInput = nil
local dragStart = nil
local startPos = nil

Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragInput = input
		dragStart = input.Position
		startPos = Main.Position
	end
end)

track(UserInputService.InputChanged:Connect(function(input)
	if not dragInput then
		return
	end
	if input == dragInput
		or (dragInput.UserInputType == Enum.UserInputType.MouseButton1
			and input.UserInputType == Enum.UserInputType.MouseMovement) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end))

track(UserInputService.InputEnded:Connect(function(input)
	if input == dragInput then
		dragInput = nil
	end
end))

local FloatingBtn = Instance.new("TextButton")
FloatingBtn.Name = "FloatingToggle"
FloatingBtn.Size = UDim2.fromOffset(48, 48)
FloatingBtn.Position = UDim2.new(0, 14, 0.5, -24)
FloatingBtn.BackgroundColor3 = Theme.Surface
FloatingBtn.Text = "◈"
FloatingBtn.TextColor3 = Theme.AccentBright
FloatingBtn.TextSize = 22
FloatingBtn.Font = Enum.Font.GothamBold
FloatingBtn.AutoButtonColor = true
FloatingBtn.ZIndex = 20
FloatingBtn.Parent = Gui

local floatCorner = Instance.new("UICorner")
floatCorner.CornerRadius = UDim.new(1, 0)
floatCorner.Parent = FloatingBtn

local floatStroke = Instance.new("UIStroke")
floatStroke.Color = Theme.Accent
floatStroke.Thickness = 1.8
floatStroke.Parent = FloatingBtn

local function toggleWindow()
	Main.Visible = not Main.Visible
	FloatingBtn.BackgroundColor3 = Main.Visible and Theme.Accent or Theme.Surface
	FloatingBtn.TextColor3 = Theme.Text
end

-- Tap opens/closes; dragging moves the floating button anywhere on screen
local btnDragInput = nil
local btnDragStart = nil
local btnStartPos = nil
local btnMoved = false

FloatingBtn.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		btnDragInput = input
		btnDragStart = input.Position
		btnStartPos = FloatingBtn.Position
		btnMoved = false
	end
end)

track(UserInputService.InputChanged:Connect(function(input)
	if not btnDragInput then
		return
	end
	if input == btnDragInput
		or (btnDragInput.UserInputType == Enum.UserInputType.MouseButton1
			and input.UserInputType == Enum.UserInputType.MouseMovement) then
		local delta = input.Position - btnDragStart
		if delta.Magnitude > 8 then
			btnMoved = true
			FloatingBtn.Position = UDim2.new(
				btnStartPos.X.Scale,
				btnStartPos.X.Offset + delta.X,
				btnStartPos.Y.Scale,
				btnStartPos.Y.Offset + delta.Y
			)
		end
	end
end))

track(UserInputService.InputEnded:Connect(function(input)
	if input == btnDragInput then
		btnDragInput = nil
		if not btnMoved then
			toggleWindow()
		end
	end
end))

CloseButton.Activated:Connect(function()
	Main.Visible = false
	FloatingBtn.BackgroundColor3 = Theme.Surface
	FloatingBtn.TextColor3 = Theme.AccentBright
end)

--========================================================
-- CLEANUP ON DESTROY
--========================================================

Gui.Destroying:Connect(function()
	for _, conn in ipairs(Connections) do
		if conn.Connected then
			conn:Disconnect()
		end
	end
	table.clear(Connections)

	for player in pairs(PlayerConnections) do
		cleanupPlayerConns(player)
	end

	for player in pairs(ESPObjects) do
		RemoveESP(player)
	end

	if ESPFolder and ESPFolder.Parent then
		ESPFolder:Destroy()
	end
end)
